ToggleButton.Position = UDim2.new(0.02, 0, 0.25, 0)

ToggleButton.Text = "MENU"

ToggleButton.Font = Enum.Font.SourceSansBold

local UICorner = Instance.new("UICorner")
UICorner.CornerRadius = UDim.new(0, 10) 
UICorner.Parent = ToggleButton

local UIStroke = Instance.new("UIStroke")
UIStroke.Color = Color3.fromRGB(0, 255, 150)
UIStroke.Thickness = 2.5
UIStroke.Parent = ToggleButton

ToggleButton.MouseButton1Click:Connect(function()
    MenuVisible = not MenuVisible
    local Success, UI = pcall(function()
        return game:GetService("CoreGui"):FindFirstChild("Fluent") or game:GetService("CoreGui"):FindFirstChild("ScreenGui", true)
    end)
    if Success and UI then UI.Enabled = MenuVisible end
end)

local UserInputService = game:GetService("UserInputService")
local dragging, dragInput, dragStart, startPos

local function update(input)
    local delta = input.Position - dragStart
    ToggleButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

ToggleButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = ToggleButton.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)

ToggleButton.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then update(input) end
end)

function Teleport(targetCFrame)
    local player = game.Players.LocalPlayer
    if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
        player.Character.HumanoidRootPart.Anchored = false
        local tweenService = game:GetService("TweenService")
        local distance = (player.Character.HumanoidRootPart.Position - targetCFrame.Position).Magnitude
        local speed = 350 
        local info = TweenInfo.new(distance / speed, Enum.EasingStyle.Linear)
        local tween = tweenService:Create(player.Character.HumanoidRootPart, info, {CFrame = targetCFrame})
        tween:Play()
        tween.Completed:Wait()
    end
end

function BringMob(mobName)
    for _, v in pairs(game:GetService("Workspace").Enemies:GetChildren()) do
        if v.Name == mobName and v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
            v.HumanoidRootPart.CFrame = game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, -5)
            v.HumanoidRootPart.CanCollide = false
            v.HumanoidRootPart.Size = Vector3.new(50, 50, 50)
        end
    end
end

function EquipWeapon()
    local player = game.Players.LocalPlayer
    if player.Character and not player.Character:FindFirstChildOfClass("Tool") then
        local tool = player.Backpack:FindFirstChild("Combat") or player.Backpack:FindFirstChildOfClass("Tool")
        if tool then player.Character.Humanoid:EquipTool(tool) end
    end
end

function Click()
    local VirtualUser = game:GetService("VirtualUser")
    VirtualUser:CaptureController()
    VirtualUser:ClickButton1(Vector2.new(1, 1))
end

function CreateESP(object, text, color)
    if object and not object:FindFirstChild("ChizHauu_ESP") then
        local billboard = Instance.new("BillboardGui")
        billboard.Name = "ChizHauu_ESP"
        billboard.AlwaysOnTop = true
        billboard.Size = UDim2.new(0, 200, 0, 50)
        billboard.StudsOffset = Vector3.new(0, 5, 0)
        billboard.Parent = object

        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 1, 0)
        label.BackgroundTransparency = 1
        label.Text = text
        label.TextColor3 = color
        label.TextSize = 16
        label.Font = Enum.Font.SourceSansBold
        label.TextStrokeTransparency = 0
        label.Parent = billboard
        
        task.spawn(function()
            while object and object.Parent and billboard and billboard.Parent do
                local pms = game.Players.LocalPlayer.Character and game.Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                if pms then
                    local dist = math.floor((object.Position - pms.Position).Magnitude)
                    label.Text = text .. " [" .. dist .. "m]"
                end
                task.wait(1)
            end
        end)
    end
end

function RemoveESP(object)
    if object and object:FindFirstChild("ChizHauu_ESP") then
        object["ChizHauu_ESP"]:Destroy()
    end
end

-- [LOGIC ĐỊNH VỊ (ESP LOOP)]
task.spawn(function()
    while task.wait(2) do
        local Mirage = game:GetService("Workspace"):FindFirstChild("Mirage Island")
        if Mirage and Mirage:IsA("Model") and getgenv().ESPMirage then
            local center = Mirage:FindFirstChild("PrimaryPart") or Mirage:FindFirstChildOfClass("BasePart")
            if center then CreateESP(center, "🐳 ĐẢO MIRAGE", Color3.fromRGB(0, 255, 255)) end
        else
            if Mirage then
                local center = Mirage:FindFirstChild("PrimaryPart") or Mirage:FindFirstChildOfClass("BasePart")
                if center then RemoveESP(center) end
            end
        end

        if getgenv().ESPGear then
            for _, v in pairs(game:GetService("Workspace"):GetChildren()) do
                if v.Name == "Gear" or v.Name == "MysteriousGear" or (v:IsA("MeshPart") and v.Size == Vector3.new(3.17, 1.25, 3.17)) then
                    CreateESP(v, "⚙️ BÁNH RĂNG V4", Color3.fromRGB(255, 215, 0))
                end
            end
        else
            for _, v in pairs(game:GetService("Workspace"):GetChildren()) do
                if v.Name == "Gear" or v.Name == "MysteriousGear" then RemoveESP(v) end
            end
        end

        if getgenv().ESPSeaMonster then
            for _, v in pairs(game:GetService("Workspace").Enemies:GetChildren()) do
                if v.Name == "Sea Beast" or v.Name == "Terrorshark" or v.Name == "Piranha" then
                    local root = v:FindFirstChild("HumanoidRootPart")
                    if root then CreateESP(root, "👹 " .. v.Name:upper(), Color3.fromRGB(255, 0, 0)) end
                end
            end
        end
    end
end)

-- Vòng lặp Auto Level (Kaitun)
task.spawn(function()
    while task.wait() do
        if getgenv().Kaitun then
            pcall(function()
                local MyLevel = game.Players.LocalPlayer.Data.Level.Value
                
                if MyLevel >= 1 and MyLevel < 15 then
                    if not game.Players.LocalPlayer.PlayerGui.Main:FindFirstChild("Quest") then
                        Teleport(CFrame.new(1059, 16, 1550)) 
                        game:GetService("ReplicatedStorage").Remotes.CommF_:InvokeServer("StartQuest", "BanditQuest1", 1)
                    else
                        local Mob = game:GetService("Workspace").Enemies:FindFirstChild("Bandit")
                        if Mob and Mob:FindFirstChild("HumanoidRootPart") then
                            Teleport(Mob.HumanoidRootPart.CFrame * CFrame.new(0, 10, 0))
                            BringMob("Bandit")
                        else
                            local Spawn = game:GetService("Workspace").EnemySpawns:FindFirstChild("Bandit")
                            if Spawn then Teleport(Spawn.CFrame) end
                        end
                    end
                
                elseif MyLevel >= 15 and MyLevel < 30 then
                    if not game.Players.LocalPlayer.PlayerGui.Main:FindFirstChild("Quest") then
                        Teleport(CFrame.new(-1598, 36, 153))
                        game:GetService("ReplicatedStorage").Remotes.CommF_:InvokeServer("StartQuest", "JungleQuest", 1)
                    else
                        local Mob = game:GetService("Workspace").Enemies:FindFirstChild("Monkey")
                        if Mob and Mob:FindFirstChild("HumanoidRootPart") then
                            Teleport(Mob.HumanoidRootPart.CFrame * CFrame.new(0, 10, 0))
                            BringMob("Monkey")
                        else
                            local Spawn = game:GetService("Workspace").EnemySpawns:FindFirstChild("Monkey")
                            if Spawn then Teleport(Spawn.CFrame) end
                        end
                    end
                else
                    MainTab:AddParagraph({Title = "Thông báo", Content = "Đã vượt mốc Level cấu hình demo!"})
                    getgenv().Kaitun = false
                end
            end)
        else
            task.wait(0.5)
        end
    end
end)

-- Vòng lặp Sea Event
task.spawn(function()
    while task.wait() do
        if getgenv().AutoSeaEvent then
            pcall(function()
                local SeaMonster = game:GetService("Workspace").Enemies:FindFirstChild("Sea Beast") 
                    or game:GetService("Workspace").Enemies:FindFirstChild("Terrorshark")
                    or game:GetService("Workspace").Enemies:FindFirstChild("Piranha")
                
                if SeaMonster and SeaMonster:FindFirstChild("HumanoidRootPart") then
                    game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = SeaMonster.HumanoidRootPart.CFrame * CFrame.new(0, 45, 0)
                    game.Players.LocalPlayer.Character.HumanoidRootPart.Anchored = true 
                    
                    local Input = game:GetService("VirtualInputManager")
                    Input:SendKeyEvent(true, Enum.KeyCode.Z, false, game) task.wait(0.05)
                    Input:SendKeyEvent(true, Enum.KeyCode.X, false, game)
                else
                    game.Players.LocalPlayer.Character.HumanoidRootPart.Anchored = false
                    local Boat = game:GetService("Workspace").Boats:FindFirstChildOfClass("Model")
                    if Boat and Boat:FindFirstChild("VehicleSeat") then
                        Teleport(Boat.VehicleSeat.CFrame)
                        Boat.VehicleSeat.Throttle = 1
                    end
                end
            end)
        else
            task.wait(0.5)
        end
    end
end)

-- Vòng lặp Auto Raid
task.spawn(function()
    while task.wait() do
        if getgenv().AutoRaid then
            pcall(function()
                if not game:GetService("Workspace"):FindFirstChild("Map") then 
                    game:GetService("ReplicatedStorage").Remotes.CommF_:InvokeServer("BlackbeardReward", "Flame", "Buy")
                    task.wait(0.5)
                    Teleport(CFrame.new(-2111, 15, -2411)) 
                    task.wait(0.5)
                    if game:GetService("Workspace").Map:FindFirstChild("RaidRoom") then
                        fireclickdetector(game:GetService("Workspace").Map.RaidRoom.Button.ClickDetector)
                    end
                else
                    local RaidMob = game:GetService("Workspace").Enemies:FindFirstChildOfClass("Model")
                    if RaidMob and RaidMob:FindFirstChild("HumanoidRootPart") and RaidMob:FindFirstChild("Humanoid") and RaidMob.Humanoid.Health > 0 then
                        Teleport(RaidMob.HumanoidRootPart.CFrame * CFrame.new(0, 12, 0))
                        BringMob(RaidMob.Name)
                    else
                        for i = 1, 5 do
                            local NextIsland = game:GetService("Workspace").Map:FindFirstChild("Island" .. i)
                            if NextIsland then
                                Teleport(NextIsland.PrimaryPart.CFrame * CFrame.new(0, 80, 0))
                                break
                            end
                        end
                    end
                end
            end)
        else
            task.wait(0.5)
        end
    end
end)

-- Vòng lặp Auto V4 nâng cao
task.spawn(function()
    while task.wait() do
        local running = false
        if getgenv().AutoMirage then
            running = true
            pcall(function()
                local Mirage = game:GetService("Workspace"):FindFirstChild("Mirage Island")
                if Mirage then Teleport(Mirage:GetModelCFrame() * CFrame.new(0, 200, 0)) end
            end)
        end
        if getgenv().AutoPullLever then
            running = true
            pcall(function()
                local Lever = game:GetService("Workspace").Map:FindFirstChild("Lever") or game:GetService("Workspace"):FindFirstChild("Lever")
                if Lever and Lever:FindFirstChild("ClickDetector") then
                    Teleport(Lever.CFrame * CFrame.new(0, 2, 0))
                    fireclickdetector(Lever.ClickDetector)
                end
            end)
        end
        if getgenv().AutoFindGear then
            running = true
            pcall(function()
                for _, v in pairs(game:GetService("Workspace"):GetChildren()) do
                    if v.Name == "Gear" or v.Name == "MysteriousGear" or (v:IsA("MeshPart") and v.Size == Vector3.new(3.17, 1.25, 3.17)) then
                        Teleport(v.CFrame * CFrame.new(0, 2, 0))
                        task.wait(0.5)
                        local vim = game:GetService("VirtualInputManager")
                        vim:SendKeyEvent(true, Enum.KeyCode.E, false, game)
                        task.wait(0.1)
                        vim:SendKeyEvent(false, Enum.KeyCode.E, false, game)
                    end
                end
            end)
        end
        if getgenv().AutoTrial then
            running = true
            pcall(function()
                local Character = game.Players.LocalPlayer.Character
                if Character and Character:FindFirstChild("Humanoid") then
                    if game.Players.LocalPlayer.PlayerGui.Main.RaceEnergy.Bar.Size.X.Scale >= 1 then
                        local vim = game:GetService("VirtualInputManager")
                        vim:SendKeyEvent(true, Enum.KeyCode.Y, false, game)
                        task.wait(0.1)
                        vim:SendKeyEvent(false, Enum.KeyCode.Y, false, game)
                    end
                end
                local TargetPlayerOrMob = game:GetService("Workspace").Enemies:FindFirstChildOfClass("Model")
                if TargetPlayerOrMob and TargetPlayerOrMob:FindFirstChild("HumanoidRootPart") then
                    Teleport(TargetPlayerOrMob.HumanoidRootPart.CFrame * CFrame.new(0, 8, 0))
                    BringMob(TargetPlayerOrMob.Name)
                end
            end)
        end
        if not running then task.wait(0.5) end
    end
end)
