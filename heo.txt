-- Code kéo thả nút Mobile
local UserInputService = game:GetService("UserInputService")
local dragging, dragInput, dragStart, startPos
local function update(input)
    local delta = input.Position - dragStart
    ToggleButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end
ToggleButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true dragStart = input.Position startPos = ToggleButton.Position
        input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
    end
end)
ToggleButton.InputChanged:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end end)
UserInputService.InputChanged:Connect(function(input) if input == dragInput and dragging then update(input) end end)

-- HỆ THỐNG QUÉT ĐỊNH VỊ HIGHLIGHT (CHẠY NGẦM)
function ApplyHighlight(target, name, color)
    local tag = target:FindFirstChild("Highlight_ESP")
    if not tag then
        local highlight = Instance.new("Highlight")
        highlight.Name = "Highlight_ESP"
        highlight.FillColor = color
        highlight.FillTransparency = 0.4
        highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
        highlight.OutlineTransparency = 0.1
        highlight.Adornee = target
        highlight.Parent = target
        
        local bill = Instance.new("BillboardGui")
        bill.Name = "Text_ESP"
        bill.AlwaysOnTop = true
        bill.Size = UDim2.new(0, 150, 0, 30)
        bill.StudsOffset = Vector3.new(0, 4, 0)
        bill.Parent = target
        
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 1, 0)
        label.BackgroundTransparency = 1
        label.Text = name
        label.TextColor3 = color
        label.TextSize = 14
        label.Font = Enum.Font.SourceSansBold
        label.TextStrokeTransparency = 0
        label.Parent = bill
    end
end

function CleanHighlight(target)
    if target:FindFirstChild("Highlight_ESP") then target["Highlight_ESP"]:Destroy() end
    if target:FindFirstChild("Text_ESP") then target["Text_ESP"]:Destroy() end
end

task.spawn(function()
    while task.wait(1) do
        local Mirage = game:GetService("Workspace"):FindFirstChild("Mirage Island")
        if Mirage and getgenv().ESPMirage then
            ApplyHighlight(Mirage, "🐳 ĐẢO MIRAGE", Color3.fromRGB(0, 235, 255))
        elseif Mirage then
            CleanHighlight(Mirage)
        end
        
        if getgenv().ESPGear then
            for _, v in pairs(game:GetService("Workspace"):GetChildren()) do
                if v.Name == "Gear" or v.Name == "MysteriousGear" or (v:IsA("MeshPart") and v.Size == Vector3.new(3.17, 1.25, 3.17)) then
                    ApplyHighlight(v, "⚙️ BÁNH RĂNG V4", Color3.fromRGB(255, 215, 0))
                end
            end
        else
            for _, v in pairs(game:GetService("Workspace"):GetChildren()) do
                if v.Name == "Gear" or v.Name == "MysteriousGear" then CleanHighlight(v) end
            end
        end

        if getgenv().ESPPlayer then
            for _, p in pairs(game.Players:GetPlayers()) do
                if p ~= game.Players.LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    ApplyHighlight(p.Character, "👤 " .. p.Name, Color3.fromRGB(0, 255, 120))
                end
            end
        else
            for _, p in pairs(game.Players:GetPlayers()) do
                if p.Character then CleanHighlight(p.Character) end
            end
        end

        if getgenv().ESPFruit then
            for _, v in pairs(game:GetService("Workspace"):GetChildren()) do
                if v:IsA("Tool") and (string.find(v.Name, "Fruit") or string.find(v.Name, "Trái")) then
                    ApplyHighlight(v, "🍓 " .. v.Name, Color3.fromRGB(255, 0, 100))
                elseif v.Name == "Fruit " or v:FindFirstChild("Fruit") then
                    ApplyHighlight(v, "🍓 Trái Ác Quỷ", Color3.fromRGB(255, 0, 100))
                end
            end
        else
            for _, v in pairs(game:GetService("Workspace"):GetChildren()) do
                if string.find(v.Name, "Fruit") or v.Name == "Fruit " then CleanHighlight(v) end
            end
        end
    end
end)

-- Vòng lặp các tính năng tự động Up V4 khác
task.spawn(function()
    while task.wait() do
        local running = false
        if getgenv().AutoMirage and game:GetService("Workspace"):FindFirstChild("Mirage Island") then
            running = true
            pcall(function() game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = game:GetService("Workspace")["Mirage Island"]:GetModelCFrame() * CFrame.new(0, 150, 0) end)
        end
        if getgenv().AutoPullLever then
            running = true
            pcall(function()
                local Lever = game:GetService("Workspace").Map:FindFirstChild("Lever") or game:GetService("Workspace"):FindFirstChild("Lever")
                if Lever and Lever:FindFirstChild("ClickDetector") then
                    game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = Lever.CFrame * CFrame.new(0, 2, 0)
                    fireclickdetector(Lever.ClickDetector)
                end
            end)
        end
        if not running then task.wait(0.5) end
    end
end)

-- LOGIC AUTO FARM BONES CHẠY NGẦM
function BringMob(mobName)
    local pms = game.Players.LocalPlayer.Character and game.Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not pms then return end
    for _, v in pairs(game:GetService("Workspace").Enemies:GetChildren()) do
        if v.Name == mobName and v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
            v.HumanoidRootPart.CFrame = pms.CFrame * CFrame.new(0, -1, -5)
            v.HumanoidRootPart.CanCollide = false
            v.HumanoidRootPart.Size = Vector3.new(10, 10, 10)
        end
    end
end

task.spawn(function()
    while task.wait() do
        if getgenv().AutoFarmBones and CurrentSea == 3 then
            pcall(function()
                local mainGui = game.Players.LocalPlayer.PlayerGui:FindFirstChild("Main")
                local hasQuest = mainGui and mainGui:FindFirstChild("Quest") and mainGui.Quest.Visible == true
                
                if not hasQuest then
                    if game.Players.LocalPlayer.Character and game.Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = CFrame.new(-9515, 160, 5520)
                        task.wait(0.5)
                        game:GetService("ReplicatedStorage").Remotes.CommF_:InvokeServer("StartQuest", "HauntedCastleQuest", 1)
                    end
                else
                    local Mob = game:GetService("Workspace").Enemies:FindFirstChild("Reborn Skeleton")
                    if Mob and Mob:FindFirstChild("HumanoidRootPart") and Mob.Humanoid.Health > 0 then
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = Mob.HumanoidRootPart.CFrame * CFrame.new(0, 11, 0)
                        BringMob("Reborn Skeleton")
                    else
                        if game.Players.LocalPlayer.Character and game.Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                            game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = CFrame.new(-9600, 165, 5600)
                        end
                    end
                end
            end)
        else
            task.wait(0.5)
        end
    end
end)
