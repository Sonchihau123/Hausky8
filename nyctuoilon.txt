local function GetQuestData()
    local lv = game.Players.LocalPlayer.Data.Level.Value
    if CurrentSea == 1 then
        if lv < 15 then return "BanditQuest1", 1, "Bandit", CFrame.new(1059, 16, 1550), CFrame.new(1145, 22, 1630)
        elseif lv < 30 then return "JungleQuest", 1, "Monkey", CFrame.new(-1598, 36, 153), CFrame.new(-1610, 40, 145)
        else return "MarineQuest", 1, "Subordinate", CFrame.new(-2439, 73, 1470), CFrame.new(-2540, 10, 1320) end
    elseif CurrentSea == 3 then
        return "HauntedCastleQuest", 1, "Reborn Skeleton", CFrame.new(-9515, 160, 5520), CFrame.new(-9600, 165, 5600)
    end
end

function BringMobsToPlayer(mobName)
    pcall(function()
        local hrp = game.Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        for _, v in pairs(game:GetService("Workspace").Enemies:GetChildren()) do
            if v.Name == mobName and v:FindFirstChild("HumanoidRootPart") and v.Humanoid.Health > 0 then
                -- FIX: Ép quái xuống thấp hẳn (-12 studs) để nhân vật đấm trúng
                v.HumanoidRootPart.CFrame = hrp.CFrame * CFrame.new(0, -12, -5)
                v.HumanoidRootPart.CanCollide = false
            end
        end
    end)
end

task.spawn(function()
    while task.wait() do
        if getgenv().Kaitun then
            pcall(function()
                local qName, qID, mobName, npcPos, mobPos = GetQuestData()
                local hasQuest = game.Players.LocalPlayer.PlayerGui.Main.Quest.Visible
                
                if not hasQuest then
                    game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = npcPos
                    game:GetService("ReplicatedStorage").Remotes.CommF_:InvokeServer("StartQuest", qName, qID)
                else
                    local TargetMob = game:GetService("Workspace").Enemies:FindFirstChild(mobName)
                    if TargetMob and TargetMob.Humanoid.Health > 0 then
                        -- Nhân vật bay cao trên quái 10 studs
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = TargetMob.HumanoidRootPart.CFrame * CFrame.new(0, 10, 0)
                        BringMobsToPlayer(mobName)
                        game:GetService("VirtualUser"):CaptureController()
                        game:GetService("VirtualUser"):ClickButton1(Vector2.new(50, 50))
                    else
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = mobPos
                    end
                end
            end)
        elseif getgenv().AutoFarmBones and CurrentSea == 3 then
             BringMobsToPlayer("Reborn Skeleton")
             game:GetService("VirtualUser"):ClickButton1(Vector2.new(50, 50))
        end
    end
end)

-- 6. HỆ THỐNG ESP HIGHLIGHT (CHẠY NGẦM)
function ApplyESP(t, c)
    if not t:FindFirstChild("Highlight_ESP") then
        local h = Instance.new("Highlight", t)
        h.Name = "Highlight_ESP"
        h.FillColor = c
        h.FillTransparency = 0.5
    end
end

task.spawn(function()
    while task.wait(2) do
        if getgenv().ESPPlayer then
            for _, p in pairs(game.Players:GetPlayers()) do
                if p ~= game.Players.LocalPlayer and p.Character then ApplyESP(p.Character, Color3.fromRGB(0, 255, 120)) end
            end
        end
    end
end)
