-- Code xử lý kéo thả nút Mobile mượt mà, không bị kẹt khi nhả tay
local UserInputService = game:GetService("UserInputService")
local dragging, dragInput, dragStart, startPos
local function update(input)
    local delta = input.Position - dragStart
    ToggleButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end
ToggleButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true dragStart = input.Position startPos = ToggleButton.Position
        input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
    end
end)
ToggleButton.InputChanged:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end end)
UserInputService.InputChanged:Connect(function(input) if input == dragInput and dragging then update(input) end end)

local function GetQuestData()
    local lv = game.Players.LocalPlayer.Data.Level.Value
    if CurrentSea == 1 then
        if lv < 15 then return "BanditQuest1", 1, "Bandit", CFrame.new(1059, 16, 1550), CFrame.new(1145, 22, 1630)
        elseif lv < 30 then return "JungleQuest", 1, "Monkey", CFrame.new(-1598, 36, 153), CFrame.new(-1610, 40, 145)
        elseif lv < 60 then return "PirateVillageQuest", 1, "Pirate", CFrame.new(-1120, 4, 3855), CFrame.new(-1195, 6, 3915)
        elseif lv < 90 then return "DesertQuest", 1, "Desert Bandit", CFrame.new(1094, 6, 4192), CFrame.new(1000, 15, 4170)
        elseif lv < 120 then return "FrozenVillageQuest", 1, "Snow Bandit", CFrame.new(1193, 6, -1157), CFrame.new(1250, 30, -1110)
        else return "MarineQuest", 1, "Subordinate", CFrame.new(-2439, 73, 1470), CFrame.new(-2540, 10, 1320) end
    elseif CurrentSea == 2 then
        if lv < 775 then return "Area1Quest", 1, "Raider", CFrame.new(-429, 73, 536), CFrame.new(-500, 75, 400)
        elseif lv < 875 then return "Area2Quest", 1, "Swan Pirate", CFrame.new(-429, 73, 536), CFrame.new(-900, 80, 750)
        else return "GreenZoneQuest", 1, "Marine Soldier", CFrame.new(-2450, 73, -2650), CFrame.new(-2300, 80, -2800) end
    elseif CurrentSea == 3 then
        if lv < 1575 then return "PortTownQuest", 1, "Pirate Millionaire", CFrame.new(-290, 70, 5300), CFrame.new(-350, 75, 5450)
        elseif lv < 1775 then return "HydraIslandQuest", 1, "Dragon Crew Warrior", CFrame.new(5210, 600, 345), CFrame.new(5400, 610, 360)
        else return "HauntedCastleQuest", 1, "Reborn Skeleton", CFrame.new(-9515, 160, 5520), CFrame.new(-9600, 165, 5600) end
    end
end

function EquipWeapon()
    pcall(function()
        local character = game.Players.LocalPlayer.Character
        local backpack = game.Players.LocalPlayer.Backpack
        if not character:FindFirstChildOfClass("Tool") then
            for _, v in pairs(backpack:GetChildren()) do
                if v:IsA("Tool") and (v.ToolTip == "Melee" or v.Name == "Combat") then
                    v.Parent = character break
                end
            end
        end
    end)
end

function BringMobsToPlayer(mobName)
    pcall(function()
        local hrp = game.Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        for _, v in pairs(game:GetService("Workspace").Enemies:GetChildren()) do
            if v.Name == mobName and v:FindFirstChild("HumanoidRootPart") and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 then
                v.HumanoidRootPart.CFrame = hrp.CFrame * CFrame.new(0, -1, -5)
                v.HumanoidRootPart.CanCollide = false
            end
        end
    end)
end

task.spawn(function()
    while task.wait() do
        if getgenv().Kaitun then
            pcall(function()
                local qName, qID, mobName, npcPos, mobPos = GetQuestData()
                local mainGui = game.Players.LocalPlayer.PlayerGui:FindFirstChild("Main")
                local hasQuest = mainGui and mainGui:FindFirstChild("Quest") and mainGui.Quest.Visible == true
                
                EquipWeapon()
                
                if not hasQuest then
                    game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = npcPos
                    task.wait(0.6)
                    game:GetService("ReplicatedStorage").Remotes.CommF_:InvokeServer("StartQuest", qName, qID)
                else
                    local TargetMob = game:GetService("Workspace").Enemies:FindFirstChild(mobName)
                    if TargetMob and TargetMob:FindFirstChild("HumanoidRootPart") and TargetMob.Humanoid.Health > 0 then
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = TargetMob.HumanoidRootPart.CFrame * CFrame.new(0, 10, 0)
                        BringMobsToPlayer(mobName)
                        local VirtualUser = game:GetService("VirtualUser")
                        VirtualUser:CaptureController()
                        VirtualUser:ClickButton1(Vector2.new(50, 50))
                    else
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = mobPos
                    end
                end
            end)
        else
            task.wait(0.5)
        end
    end
end)
